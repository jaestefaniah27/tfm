% --- Figuras aún no disponibles ---------------------------------------------
% \figpendiente{ruta}{ancho}{descripción}: si el fichero existe lo inserta como
% una figura normal; si todavía no está, dibuja un recuadro con la descripción y
% la ruta esperada, de modo que el documento siga compilando. Basta con dejar el
% fichero en su sitio para que la figura aparezca sin tocar el .tex.
\newcommand{\figpendiente}[3]{%
  \IfFileExists{#1}%
    {\includegraphics[width=#2]{#1}}%
    {\fbox{\parbox{0.92\textwidth}{\centering\vspace{0.45cm}%
       \textit{[Figura pendiente: #3]}\\[0.25cm]%
       {\small\texttt{\detokenize{#1}}}\vspace{0.45cm}}}}%
}
